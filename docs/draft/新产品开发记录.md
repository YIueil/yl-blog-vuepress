```sql
select spaceanaly0_.ID                  as col_0_0_,  
       spaceanaly0_.BUSINESS_ID_        as BUSINESS_ID_,  
       spaceanaly0_.ITEM_ID_            as col_2_0_,  
       spaceanaly1_.ITEM_CODE_          as col_3_0_,

         left join ynyth_inst.app_sa_business asb on spaceanaly0_.business_id_ = asb.business_id_  
where spaceanaly0_.ENABLED_=1  
order by spaceanaly0_.ORDER_NO_ asc;
```
